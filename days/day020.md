    orange.parentElement.style.padding = "20px";
    orange.parentElement.style.borderRadius = "20px";

    //8. 제일가까운 상위부모 ( id.closest("id를 감싸는 태그") )

    // h4는 orange를 감싸고 있지 않아서 불가능 (div, ul 가능)
    orange.closest("div").style.color = "gold";
