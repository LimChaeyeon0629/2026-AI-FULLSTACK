      window.addEventListener("load", function() {
        //////////////////////////////
        //Q1) document.getElementsById 이용해서  #fruits 선택
        let fru = document.getElementById("fruits");  
        console.log(fru);

        //Q2) document.getElementsByTagName 이용해서
        //   #fruits 안의 모든 li에 배경색 black, 글자색 white , 여백 10px주기
        //   ★ getElementsByTagName = 여러 요소 선택 가능, 태그 이름, for문 사용
        //   → fruits.getElements.ByTagName("li") → fruits 안에 "li" 선택자
        //   → 변수명 지정 → for문 사용 → 변수명[i].style.bgColor
        let frus = fruits.getElementsByTagName("li");
        for(i=0; i<frus.length; i++) {
          frus[i].style.backgroundColor = "black";
          frus[i].style.color = "white";
          frus[i].style.padding = "10px";
        }
        //Q2) document.getElementsById 이용해서 banana 선택 배경색 gold
        //    ★ getElementsById = 단일 요소 선택, id 속성값
        //    → document.getElementsById("banana"); → 변수명 지정 → 변수명.style.bgColor
        let frus1 = document.getElementById("banana");
        frus1.style.backgroundColor = "gold";
        // this.style.backgroundColor = "gold";

        //Q3) document.querySelector 이용해서 coconut 선택
        //  coconut를 클릭하면 width:200px; 로 수정
        //  ★ (css id="coconut") → "#coconut"
        document.querySelector("#coconut").onclick = function() {
          this.style.width = "200px";
        };
      });
    </script>
